} // 元のメッセージID → { content, quote, channelId, records, reactions } // （削除・編集・リアクションの連動用） const forwarded = new Map(); // 元・転送先どちらのメッセージIDからでも、元のIDを引けるようにする const groupIndex = new Map(); function remember(originId, entry) { forwarded.set(originId, entry); if (!entry) return; groupIndex.delete(originId); for (const r of entry.records) groupIndex.delete(r.id); forwarded.delete(originId); } // 誰が何を押したかの行（転送先メッセージの下に付ける） function reactionLine(entry) { if (!entry.reactions || entry.reactions.size === 0) return null; const parts = [...entry.reactions.values()].map((r) => { const names = [...r.names.values()]; const shown = names.slice(0, 8).join(', '); const rest = names.length > 8 ? ` 他${names.length - 8}人` : ''; return `${r.emoji} ${shown}${rest}`; }); return '-# ' + parts.join(' ／ '); } function withReactions(entry, base) { const line = reactionLine(entry); if (!line) return base; const room = 2000 - line.length - 1; return [base?.slice(0, room), line].filter(Boolean).join('\n'); } // 転送先のメッセージ全部の「押した人の行」を書き換える async function refreshReactionLines(entry) { await Promise.allSettled( entry.records.map(async (rec) => { try { const webhook = new WebhookClient({ id: rec.info.webhookId, token: rec.info.webhookToken, }); await webhook.editMessage(rec.id, { content: withReactions(entry, rec.content) ?? '', allowedMentions: { parse: [] }, }); } catch (err) { console.error('リアクション表示の更新に失敗:', err.message); } }) ); } async function displayName(message, user) { try { const member = await message.guild.members.fetch(user.id); return member.displayName; } catch { return user.username ?? '???'; } } // 言語ごとに1回だけ翻訳する関数を作る function makeTranslateTo(content, canTranslate) { const cache = {}; return async function translateTo(lang) { if (!content || !lang || !canTranslate) return content; if (cache[lang] !== undefined) return cache[lang]; try { const result = await translator.translateText(content, null, lang); cache[lang] = result.text; } catch (err) { console.error('翻訳失敗:', err.message); cache[lang] = content; } return cache[lang]; }; } // 返信先の引用（名前と本文の冒頭）を作る async function buildQuote(message) { if (!message.reference?.messageId) return null; try { const ref = await message.fetchReference(); const name = (ref.member?.displayName ?? ref.author.username).slice(0, 40); let snippet = ref.content .split('\n') .filter((l) => !l.startsWith('> ') && !l.startsWith('-# ')) .join(' ') .trim(); if (snippet.length > 60) snippet = snippet.slice(0, 60) + '…'; if (!snippet) snippet = ref.attachments.size > 0 ? '📎' : ''; return snippet ? { name, snippet } : null; } catch { return null; } } // 転送先に送る本文を作る async function render(message, info, translateTo, canTranslate, quote, quoteTranslateTo) { const text = await translateTo(info.language); let quoteLine = null; if (quote) { const q = (await quoteTranslateTo(info.language)).replace(/\n/g, ' '); quoteLine = `> ↩ **${quote.name}**: ${q}`; } const notice = !canTranslate && message.content && info.language ? '-# ⚠️ 翻訳の上限に達したため、原文のまま転送しています' : null; const attachmentUrls = [...message.attachments.values()].map((a) => a.url); const body = [quoteLine, text, notice, ...attachmentUrls].filter(Boolean).join('\n'); return body ? body.slice(0, 2000) : null; } const client = new Client({ intents: [ GatewayIntentBits.Guilds, GatewayIntentBits.GuildMessages, GatewayIntentBits.MessageContent, GatewayIntentBits.GuildWebhooks, GatewayIntentBits.GuildMessageReactions, ], partials: [Partials.Message, Partials.Channel, Partials.Reaction], }); } catch (err) { console.error('コマンド登録失敗:', err.message); } }); client.on(Events.InteractionCreate, async (interaction) => { if (!interaction.isChatInputCommand()) return; const command = client.commands.get(interaction.commandName); } catch (err) { console.error(`転送失敗 → ${channelId}:`, err.message); } }) ); if (records.length > 0) { remember(message.id, { content: message.content, quote, channelId: message.channelId, records, reactions: new Map(), }); } }); // 元のメッセージが編集されたら、転送先も書き換える client.on(Events.MessageUpdate, async (oldMessage, newMessage) => { const entry = forwarded.get(newMessage.id); if (!entry) return; if (newMessage.partial) { try { newMessage = await newMessage.fetch(); } catch { return; } } // 本文が変わっていない更新（リンクのプレビュー表示など）は無視 if (newMessage.content === entry.content) return; entry.content = newMessage.content; const canTranslate = await translationAvailable(); const translateTo = makeTranslateTo(newMessage.content, canTranslate); if (!finalContent) return; rec.content = finalContent; const webhook = new WebhookClient({ id: rec.info.webhookId, token: rec.info.webhookToken, }); await webhook.editMessage(rec.id, { content: withReactions(entry, finalContent), allowedMentions: { parse: [] }, }); } catch (err) { console.error('転送先の編集に失敗:', err.message); } }) ); }); // 元のメッセージが消されたら、転送先も消す client.on(Events.MessageDelete, async (message) => { const entry = forwarded.get(message.id); if (!entry) return; forget(message.id); await Promise.allSettled( entry.records.map(async ({ info, id }) => { try { const webhook = new WebhookClient({ id: info.webhookId, token: info.webhookToken, }); await webhook.deleteMessage(id); } catch (err) { console.error('転送先の削除に失敗:', err.message); } }) ); }); // リアクションが付いたら、他のコピーにも付けて、押した人の行を更新する client.on(Events.MessageReactionAdd, async (reaction, user) => { if (user.bot) return; const originId = groupIndex.get(reaction.message.id);
